[pi5]
# KMS DPI display path for Raspberry Pi 5 (RP1). Same pins as the Pi 4 path:
# VSYNC GPIO2, HSYNC GPIO3, RGB666 on GPIO4-21, both syncs active high.
# Timings match the Pi 4 hdmi_timings above (7.3632 MHz, 376x276 @ 50.56 Hz).
# The Pi 5 has no analog audio; the HAT needs a USB sound card for audio.
# config.txt lines are cut at 98 characters, so the timings go on dtparam lines.

dtparam=hactive=376,hfp=16,hsync=32,hbp=55
dtparam=vactive=276,vfp=3,vsync=2,vbp=23
